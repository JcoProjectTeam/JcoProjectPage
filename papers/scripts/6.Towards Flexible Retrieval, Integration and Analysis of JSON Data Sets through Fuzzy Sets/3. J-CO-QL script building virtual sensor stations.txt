//12
JOIN OF COLLECTIONS
		NitricOxide_Sensors AS NO, Weather_Sensors AS WS
	ON GEOMETRY DISTANCE(KM) < 2.0
	SET GEOMETRY LEFT
	CASES
		WHERE	
				.WS.sensorType = "Global Radiation" 
			GENERATE	
				KEEPING GEOMETRY
				BUILD	{
					.GRsensorId: .WS.sensorId, 
					.NOsensorId: .NO.sensorId, 
					.NOstationName: .NO.stationName	}
		DROP OTHERS;

//13
SAVE AS SensorPairs;

//14
JOIN OF COLLECTIONS
		SensorPairs AS SP, Weather_Sensors AS WS
	ON GEOMETRY DISTANCE(KM) < 2.0
	SET GEOMETRY LEFT
	CASES
		WHERE	
				.WS.sensorType = "Rain Precipitation" 
			GENERATE	
				KEEPING GEOMETRY
				BUILD	{
					.RPsensorId: .WS.sensorId, 
					.GRsensorId: .SP.GRsensorId, 
					.NOsensorId: .SP.NOsensorId, 
					.NOstationName: .SP.NOstationName	}
		DROP OTHERS;

//15
SAVE AS SensorTriplets;
