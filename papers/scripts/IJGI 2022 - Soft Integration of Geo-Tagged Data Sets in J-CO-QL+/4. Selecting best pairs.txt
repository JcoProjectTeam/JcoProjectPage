GET COLLECTION RelevantPairs@ijgiDb;

GROUP
    PARTITION   WITH .g.gId
        BY .g.gId
        INTO .gGroup
            ORDER BY .rank TYPE NUMERIC DESC;

EXPAND
    UNPACK WITH .gGroup
        ARRAY .gGroup
        TO .gPair;

FILTER
    CASE WHERE WITH .gPair AND
                    .gPair.position = 1
        GENERATE
            BUILD {
                .f : .gPair.item.f,
                .g : .gPair.item.g,
                .rank : .gPair.item.rank };

SAVE AS SamePlaces@ijgiDb;
