//2.
USE DB Boundaries
	ON SERVER jcods 'http://10.0.0.11:17017';

//3.
USE DB RegionInfo
	ON SERVER MongoDB 'http://10.0.0.12:27017';

//4.
USE DB ToShare
	ON SERVER MongoDB 'http://10.0.0.13:27017';

//5.
GET COLLECTION Districts@Boundaries;

//6.
FILTER 
	CASES
		WHERE	
					WITH .Properties.City
				AND	
					.Properties.City = "Milan"  
			GENERATE	
				KEEPING GEOMETRY
				BUILD	{
					.ID: .ID, 
					.Name: .Properties.Name	}
		DROP OTHERS;

//7.
SAVE AS Milan_Districts;

//8.
GET COLLECTION BikeLanes@RegionInfo;

//9.
FILTER 
	CASES
		WHERE	
					WITH .City
				AND	
					.City = "Milan"  
			GENERATE	
				SETTING GEOMETRY .geometry
				BUILD	{
					.Name: .Name	}
		WHERE	
					WITH .Properties.City
				AND	
					.Properties.City = "Milan"  
			GENERATE	
				SETTING GEOMETRY .geometry
				BUILD	{
					.Name: .Properties.Name	}
		DROP OTHERS;

//10.
SAVE AS Milan_BikeLanes;
