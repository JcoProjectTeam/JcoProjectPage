//10: 
JOIN OF COLLECTIONS
		Milan_Districts AS D, Milan_BikeLanes AS B
	ON GEOMETRY INTERSECT
	SET GEOMETRY INTERSECTION
	CASES
		WHERE	
				WITH .D.ID, .B.Name 
			GENERATE	
				KEEPING GEOMETRY
				BUILD	{
					.DistrictName: .D.Name, 
					.BikeLaneName: .B.Name	}
		DROP OTHERS
	REMOVE DUPLICATES;

//11.
GROUP
	PARTITION WITH .BikeLaneName
		BY .BikeLaneName
		INTO .DistrictSegs
		DROP GROUPING FIELDS
	DROP OTHERS;

//12.
FILTER 
	CASES
		WHERE	
				WITH .BikeLaneName, .DistrictSegs 
			GENERATE	
				SETTING GEOMETRY AGGREGATE (.DistrictSegs)
				BUILD	{
					.BikeLaneName: .BikeLaneName, 
					.NumOfDistricts: COUNT(.DistrictSegs)	}
		DROP OTHERS;

//13.
FILTER 
	CASES
		WHERE	
					WITH .BikeLaneName, .NumOfDistricts
				AND	
					.NumOfDistricts >= 2  
			GENERATE	
				KEEPING GEOMETRY
				BUILD	{
					.BikeLaneName: .BikeLaneName	}
		DROP OTHERS;

//14.
SAVE AS BikeLanesCrossingDistricts@ToShare;
