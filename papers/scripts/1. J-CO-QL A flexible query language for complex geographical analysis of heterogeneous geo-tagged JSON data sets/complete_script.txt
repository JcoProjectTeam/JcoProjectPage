JOIN OF COLLECTIONS
		Buildings@ToyDB AS B, Restaurants@ToyDB AS R
	CASES
		WHERE	
					WITH STRING .B.City, .B.Address, .R.City, .R.Address
				AND	
					WITH GEOMETRY .R.geometry
				AND	
					.B.City = .R.City
				AND	
					.B.Address = .R.Address  
			GENERATE	
				SETTING GEOMETRY .B.geometry
				BUILD	{
					.BuildingName: .B.Name, 
					.RestaurantName: .R.Name, 
					.City: .B.City, 
					.Address: .B.Address	}
		WHERE	
					WITH STRING .B.CityName, .B.Address, .R.City, .R.Address
				AND	
					WITH GEOMETRY .R.geometry
				AND	
					.B.CityName = .R.City
				AND	
					.B.Address = .R.Address  
			GENERATE	
				SETTING GEOMETRY .B.geometry
				BUILD	{
					.BuildingName: .B.Name, 
					.RestaurantName: .R.Name, 
					.City: .B.CityName, 
					.Address: .B.Address	}
		DROP OTHERS;


GET COLLECTION Restaurants@ToyDB;


FILTER 
	CASES
		WHERE	
					WITH .lat, .lng
				AND	
					WITHOUT .geometry  
			GENERATE	
				SETTING GEOMETRY POINT (.lat, .lng)
		KEEP OTHERS;


GET COLLECTION WaterLines@ToyDB;


GROUP
	PARTITION WITH STRING .city
		BY .city
		INTO .waterLineCity
	DROP OTHERS;


GET COLLECTION WaterLines@ToyDB;


GROUP
	PARTITION WITH STRING .city
		BY .city
		INTO .waterLineCity 
		GENERATE	
			SETTING GEOMETRY AGGREGATE (.waterLineCity)
	DROP OTHERS;


GROUP
	PARTITION WITH .RailwayOwner, .Type, .maxSpeed
		BY .RailwayOwner, .Type
		INTO .groupedRailways
		SORTED BY .maxSpeed TYPE NUMERIC
	PARTITION  WITH STRING .countyName, .townName AND WITH FLOAT .width 
		BY .countyName, .townName
		INTO .groupedRoads
	DROP OTHERS;


EXPAND
	UNPACK
				WITH STRING .city
			AND	
				WITH ARRAY .waterLineCity 
		ARRAY .waterLineCity
		TO .tmp
	DROP OTHERS;


FILTER 
	CASES
		WHERE	
				WITH ARRAY .waterLineCity 
			GENERATE	
				SETTING GEOMETRY .tmp.geometry
				BUILD	{
					.tmp.name: .tmp.name, 
					.tmp.city: .tmp.city	};
