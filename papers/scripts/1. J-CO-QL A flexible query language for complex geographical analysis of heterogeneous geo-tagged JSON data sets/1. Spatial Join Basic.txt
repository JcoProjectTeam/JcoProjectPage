JOIN OF COLLECTIONS
		Buildings@ToyDB, WaterLines@ToyDB
	ON GEOMETRY INTERSECT
	SET GEOMETRY INTERSECTION
	CASES
		WHERE	
					WITH STRING .Buildings.City
				AND	
					.Buildings.City = "City A" 
		WHERE	
					WITH STRING .Buildings.CityName
				AND	
					.Buildings.CityName = "City A" 
		DROP OTHERS;
