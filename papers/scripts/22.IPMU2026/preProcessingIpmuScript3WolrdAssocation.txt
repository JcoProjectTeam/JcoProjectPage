
/*
x.	Connecting to LGrade database
*/
USE DB ipmuDbPrep
	ON SERVER MongoDB 'http://127.0.0.1:27020';


/*
	building depth 2 connections
*/

JOIN OF COLLECTIONS	b40WorldExpanded@ipmuDbPrep AS anc,
					b40WorldExpanded@ipmuDbPrep AS desc
	CASES
		WHERE
				.anc.placeId = .desc.ancestorId AND
				WITH .anc.ancestorId
			GENERATE
				BUILD {
					.placeId: 			.desc.placeId, 
					.name: 				.desc.name, 
					.depth: 			.desc.depth, 
					.granularity: 		.desc.granularity, 
					.children:			.desc.children, 
					.nChildren: 		COUNT(.desc.children), 				
					.ancestorId: 		.anc.ancestorId, 
					.distance2ancestor: 2
				};
			
SAVE AS b41WorldDepth2Temp@ipmuDbPrep;

GROUP PARTITION WITH .ancestorId
	BY .ancestorId
	INTO .ancestorHeirs;

SAVE AS b41WorldDepth2Grouped@ipmuDbPrep;

JOIN OF COLLECTIONS	b41WorldDepth2Temp@ipmuDbPrep AS loc,
					b41WorldDepth2Grouped@ipmuDbPrep AS group
	CASES WHERE .loc.ancestorId = .group.ancestorId
	GENERATE BUILD{
					.placeId: 			.loc.placeId, 
					.name: 				.loc.name, 
					.depth: 			.loc.depth, 
					.granularity: 		.loc.granularity, 
					.children: 			.loc.children, 
					.nChildren: 		.loc.nChildren,
					.ancestorId: 		.loc.ancestorId, 
					.distance2ancestor:	.loc.distance2ancestor, 
					.ancestorHeirs: 	COUNT(.group.ancestorHeirs)
	};

SAVE AS b41WorldDepth2@ipmuDbPrep;
